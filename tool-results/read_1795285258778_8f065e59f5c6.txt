    24→    24→/** Keep only 10-digit Indian mobile numbers (strips +91 / 91 prefixes). */
    25→    25→export function normalizePhone(phone?: string | null): string | null {
    26→    26→  if (!phone) return null;
    27→    27→  let digits = String(phone).replace(/\D/g, "");
    28→    28→  if (digits.startsWith("91") && digits.length === 12) digits = digits.slice(2);
    29→    29→  if (digits.length === 10) return digits;
    30→    30→  return null;
    31→    31→}

   170→   170→export async function addTransaction(
   171→   171→  input: AddTxnInput
   172→   172→): Promise<{ transaction: TxnPublic; customer: CustomerPublic; duplicate?: TxnPublic }> {
   173→   173→  if (!Number.isInteger(input.amountPaise) || input.amountPaise <= 0) {
   174→   174→    throw new Error("Amount must be greater than zero.");
   175→   175→  }
   176→   176→  const key = nameKey(input.customer);
   177→   177→  if (!key) throw new Error("Customer name is required.");
   178→   178→  const phone = normalizePhone(input.phone);
   179→   179→  if (input.phone && !phone) {
   180→   180→    throw new Error("Phone must be a 10-digit Indian mobile number (with optional +91).");
   181→   181→  }
   182→   182→
   183→   183→  const existing = await db.customer.findUnique({ where: { nameKey: key } });
   184→   184→
   185→   185→  if (!input.force && !input.isReversal) {
   186→   186→    const dupe = await db.transaction.findFirst({
   187→   187→      where: {
   188→   188→        nameKey: key,
   189→   189→        txnType: input.txnType,
   190→   190→        amountPaise: input.amountPaise,
   191→   191→        reversed: false,
   192→   192→        isReversal: false,
   193→   193→        createdAt: { gte: new Date(Date.now() - DUPLICATE_WINDOW_MS) },
   194→   194→      },
   195→   195→      orderBy: { createdAt: "desc" },
   196→   196→    });
   197→   197→    if (dupe) {
   198→   198→      return {
   199→   199→        transaction: publicTxn(dupe),
   200→   200→        customer: existing ? publicCustomer(existing) : publicTxnAsCustomer(dupe),
   201→   201→        duplicate: publicTxn(dupe),
   202→   202→      };
   203→   203→    }
   204→   204→  }
   205→   205→
   206→   206→  const now = new Date();
   207→   207→  const delta = input.txnType === "UDHAAR" ? input.amountPaise : -input.amountPaise;
   208→   208→  const oldBalance = existing?.balancePaise ?? 0;
   209→   209→  const newBalance = oldBalance + delta;
   210→   210→
   211→   211→  let firstDueAt = existing?.firstDueAt ?? null;
   212→   212→  if (newBalance > 0 && (!firstDueAt || oldBalance <= 0)) firstDueAt = now;
   213→   213→  if (newBalance <= 0) firstDueAt = null;
   214→   214→
   215→   215→  const displayName = existing?.name ?? input.customer.trim();
   216→   216→  const savedPhone = phone ?? existing?.phone ?? null;
   217→   217→
   218→   218→  const txn = await db.$transaction(async (tx) => {
   219→   219→    await tx.customer.upsert({
   220→   220→      where: { nameKey: key },
   221→   221→      create: {
   222→   222→        nameKey: key,
   223→   223→        name: displayName,
   224→   224→        balancePaise: newBalance,
   225→   225→        phone: savedPhone,
   226→   226→        firstDueAt,
   227→   227→        lastTxnAt: now,
   228→   228→      },
   229→   229→      update: {
   230→   230→        name: displayName,
   231→   231→        balancePaise: newBalance,
   232→   232→        phone: savedPhone,
   233→   233→        firstDueAt,
   234→   234→        lastTxnAt: now,
   235→   235→      },
   236→   236→    });
   237→   237→    const created = await tx.transaction.create({
   238→   238→      data: {
   239→   239→        nameKey: key,
   240→   240→        name: displayName,
   241→   241→        amountPaise: input.amountPaise,
   242→   242→        txnType: input.txnType,
   243→   243→        rawText: input.rawText,
   244→   244→        note: input.note ?? null,
   245→   245→        isReversal: input.isReversal ?? false,
   246→   246→        reversedById: input.reversedById ?? null,
   247→   247→        createdAt: now,
   248→   248→      },
   249→   249→    });
   250→   250→    await tx.auditEvent.create({
   251→   251→      data: {
   252→   252→        action: "transaction_create",
   253→   253→        entity: "transaction",
   254→   254→        entityId: created.id,
   255→   255→        data: JSON.stringify({
   256→   256→          customer: displayName,
   257→   257→          txnType: input.txnType,
   258→   258→          amountPaise: input.amountPaise,
   259→   259→          forced: input.force ?? false,
   260→   260→          isReversal: input.isReversal ?? false,
   261→   261→        }),
   262→   262→      },
   263→   263→    });
   264→   264→    return created;
   265→   265→  });
   266→   266→
   267→   267→  const cust = (await db.customer.findUnique({ where: { nameKey: key } }))!;
   268→   268→  return { transaction: publicTxn(txn), customer: publicCustomer(cust) };
   269→   269→}

   285→   285→/**
   286→   286→ * Reverse a ledger line by creating an offsetting entry. The original is
   287→   287→ * flagged (never deleted or mutated in place) so the audit history survives.
   288→   288→ * Reversal lines are excluded from report totals — no money actually moved.
   289→   289→ */
   290→   290→export async function reverseTransaction(id: string) {
   291→   291→  const result = await db.$transaction(async (tx) => {
   292→   292→    const orig = await tx.transaction.findUnique({ where: { id } });
   293→   293→    if (!orig) throw new Error("Transaction not found.");
   294→   294→    if (orig.reversed) throw new Error("This transaction has already been reversed.");
   295→   295→    if (orig.isReversal) {
   296→   296→      throw new Error("A reversal entry cannot be reversed. Reverse the original transaction instead.");
   297→   297→    }
   298→   298→    const customer = await tx.customer.findUnique({ where: { nameKey: orig.nameKey } });
   299→   299→    if (!customer) throw new Error("Customer record missing for this transaction.");
   300→   300→
   301→   301→    const now = new Date();
   302→   302→    const flip: TxnType = orig.txnType === "UDHAAR" ? "PAYMENT" : "UDHAAR";
   303→   303→    const delta = flip === "UDHAAR" ? orig.amountPaise : -orig.amountPaise;
   304→   304→    const newBalance = customer.balancePaise + delta;
   305→   305→
   306→   306→    let firstDueAt = customer.firstDueAt;
   307→   307→    if (newBalance > 0 && (!firstDueAt || customer.balancePaise <= 0)) firstDueAt = now;
   308→   308→    if (newBalance <= 0) firstDueAt = null;
   309→   309→
   310→   310→    const reversal = await tx.transaction.create({
   311→   311→      data: {
   312→   312→        nameKey: orig.nameKey,
   313→   313→        name: orig.name,
   314→   314→        amountPaise: orig.amountPaise,
   315→   315→        txnType: flip,
   316→   316→        rawText: `Reversal of: ${orig.rawText}`,
   317→   317→        note: `Reversal of transaction ${orig.id}`,
   318→   318→        isReversal: true,
   319→   319→        reversedById: orig.id,
   320→   320→        createdAt: now,
   321→   321→      },
   322→   322→    });
   323→   323→    await tx.transaction.update({
   324→   324→      where: { id: orig.id },
   325→   325→      data: { reversed: true, reversedById: reversal.id },
   326→   326→    });
   327→   327→    const updated = await tx.customer.update({
   328→   328→      where: { nameKey: orig.nameKey },
   329→   329→      data: { balancePaise: newBalance, firstDueAt, lastTxnAt: now },
   330→   330→    });
   331→   331→    await tx.auditEvent.create({
   332→   332→      data: {
   333→   333→        action: "transaction_reverse",
   334→   334→        entity: "transaction",
   335→   335→        entityId: orig.id,
   336→   336→        data: JSON.stringify({
   337→   337→          reversalId: reversal.id,
   338→   338→          customer: orig.name,
   339→   339→          amountPaise: orig.amountPaise,
   340→   340→          originalType: orig.txnType,
   341→   341→        }),
   342→   342→      },
   343→   343→    });
   344→   344→    return { reversal, customer: updated };
   345→   345→  });
   346→   346→  return {
   347→   347→    transaction: publicTxn(result.reversal),
   348→   348→    customer: publicCustomer(result.customer),
   349→   349→  };
   350→   350→}

   405→   405→        data: JSON.stringify({
   406→   406→          phoneChanged: patch.phone !== undefined,
   407→   407→          notesChanged: patch.notes !== undefined,
   408→   408→        }),
   409→   409→      },
   410→   410→    });
   411→   411→    return row;
   412→   412→  });

   448→   448→/**
   449→   449→ * Reminder run (the daily-job equivalent): find customers whose dues are
   450→   450→ * older than `overdueDays` (default from Settings) and write reminder records
   451→   451→ * with ready WhatsApp links.
   452→   452→ */
   453→   453→export async function writeReminders(overdueDays?: number): Promise<ReminderPublic[]> {
   454→   454→  const settings = await getSettings();
   455→   455→  const days = Math.max(0, Math.floor(overdueDays ?? settings.overdueDays));
   456→   456→  const rid = Math.random().toString(36).slice(2, 12);
   457→   457→  const written: ReminderPublic[] = [];
   458→   458→
   459→   459→  for (const c of await db.customer.findMany()) {
   460→   460→    if (c.balancePaise <= 0) continue;
   461→   461→    const dueFrom = c.firstDueAt ?? c.lastTxnAt;
   462→   462→    const age = daysSince(dueFrom);
   463→   463→    if (age < days) continue;
   464→   464→
   465→   465→    const { url, message } = await whatsappLink(c.name, c.balancePaise, c.phone, age);
   466→   466→    const row = await db.reminder.create({
   467→   467→      data: {
   468→   468→        nameKey: c.nameKey,
   469→   469→        name: c.name,
   470→   470→        balancePaise: c.balancePaise,
   471→   471→        daysOverdue: age,
   472→   472→        phone: c.phone,
   473→   473→        whatsappUrl: url,
   474→   474→        message,
   475→   475→        runId: rid,
   476→   476→      },
   477→   477→    });
   478→   478→    written.push(publicReminder(row));
   479→   479→  }
   480→   480→  return written;
   481→   481→}

   484→   484→export async function customerReminder(name: string): Promise<ReminderPublic | null> {
   485→   485→  const key = nameKey(name);
   486→   486→  const c = await db.customer.findUnique({ where: { nameKey: key } });
   487→   487→  if (!c) return null;
   488→   488→  if (c.balancePaise <= 0) throw new Error(`${c.name} has no outstanding balance.`);

   561→   561→/* ---------------- reports ---------------- */
   562→   562→
   563→   563→export type ReportRange = "today" | "7d" | "30d" | "all" | "custom";
   564→   564→
   565→   565→/**
   566→   566→ * Reports computed from actual stored transactions.
   567→   567→ *
   568→   568→ * Counting rules (documented, deterministic, no double-counting):
   569→   569→ * - Lines with `reversed=true` are voided → excluded from totals.
   570→   570→ * - Lines with `isReversal=true` are corrections, not real money movement →
   571→   571→ *   excluded from totals (but visible in recent activity).
   572→   572→ * - Current outstanding comes from Customer.balancePaise (source of truth).
   573→   573→ * - Day boundaries are in IST (Asia/Kolkata).
   574→   574→ */
   575→   575→export async function getReports(
   576→   576→  range: ReportRange,
   577→   577→  fromStr?: string,
   578→   578→  toStr?: string
   579→   579→) {
   580→   580→  const now = new Date();
   581→   581→  let fromD: Date;
   582→   582→  let toD = now;
   583→   583→  let label: string;
   584→   584→
   585→   585→  switch (range) {
   586→   586→    case "today":
   587→   587→      fromD = istDayStart(now);
   588→   588→      label = "Today";
   589→   589→      break;
   590→   590→    case "7d":
   591→   591→      fromD = istDayStart(new Date(now.getTime() - 6 * 86_400_000));
   592→   592→      label = "Last 7 days";
   593→   593→      break;
   594→   594→    case "30d":
   595→   595→      fromD = istDayStart(new Date(now.getTime() - 29 * 86_400_000));
   596→   596→      label = "Last 30 days";
   597→   597→      break;
   598→   598→    case "custom": {
   599→   599→      if (!fromStr || Number.isNaN(new Date(fromStr).getTime())) {
   600→   600→        throw new Error("A valid 'from' date is required for a custom range.");
   601→   601→      }
   602→   602→      fromD = new Date(fromStr);
   603→   603→      if (toStr && !Number.isNaN(new Date(toStr).getTime())) toD = new Date(toStr);
   604→   604→      label = "Custom range";
   605→   605→      break;
   606→   606→    }
   607→   607→    case "all":
   608→   608→    default:
   609→   609→      fromD = new Date(0);
   610→   610→      label = "All time";
   611→   611→      break;
   612→   612→  }
   613→   613→
   614→   614→  const txns = await db.transaction.findMany({
   615→   615→    where: { createdAt: { gte: fromD, lte: toD } },
   616→   616→    orderBy: { createdAt: "desc" },
   617→   617→  });
   618→   618→  const countable = txns.filter((t) => !t.reversed && !t.isReversal);
   619→   619→
   620→   620→  let creditPaise = 0;
   621→   621→  let paymentsPaise = 0;
   622→   622→  const seriesMap = new Map<string, { credit: number; payments: number }>();
   623→   623→  for (const t of countable) {
   624→   624→    const key = istDayKey(t.createdAt);
   625→   625→    const bucket = seriesMap.get(key) ?? { credit: 0, payments: 0 };
   626→   626→    if (t.txnType === "UDHAAR") {
   627→   627→      creditPaise += t.amountPaise;
   628→   628→      bucket.credit += t.amountPaise;
   629→   629→    } else {
   630→   630→      paymentsPaise += t.amountPaise;
   631→   631→      bucket.payments += t.amountPaise;
   632→   632→    }
   633→   633→    seriesMap.set(key, bucket);
   634→   634→  }
   635→   635→
   636→   636→  // Fill day buckets for short ranges so charts show continuous days.
   637→   637→  const series: { date: string; credit: number; payments: number }[] = [];
   638→   638→  const fillFrom = new Date(Math.max(fromD.getTime(), now.getTime() - 30 * 86_400_000));
   639→   639→  if (range !== "all") {
   640→   640→    for (let d = istDayStart(fillFrom); d.getTime() <= toD.getTime(); d = new Date(d.getTime() + 86_400_000)) {
   641→   641→      const key = istDayKey(d);
   642→   642→      const b = seriesMap.get(key) ?? { credit: 0, payments: 0 };
   643→   643→      series.push({ date: key, credit: b.credit / 100, payments: b.payments / 100 });
   644→   644→    }
   645→   645→  } else {
   646→   646→    for (const key of [...seriesMap.keys()].sort()) {
   647→   647→      const b = seriesMap.get(key)!;
   648→   648→      series.push({ date: key, credit: b.credit / 100, payments: b.payments / 100 });
   649→   649→    }
   650→   650→  }
   651→   651→
   652→   652→  const customers = await db.customer.findMany();
   653→   653→  const customerOutstanding = customers
   654→   654→    .filter((c) => c.balancePaise > 0)
   655→   655→    .map((c) => ({
   656→   656→      name: c.name,
   657→   657→      phone: c.phone,
   658→   658→      balance: c.balancePaise / 100,
   659→   659→      balancePaise: c.balancePaise,
   660→   660→      daysDue: daysSince(c.firstDueAt ?? c.lastTxnAt),
   661→   661→      notes: c.notes,
   662→   662→    }))
   663→   663→    .sort((a, b) => b.balancePaise - a.balancePaise);
   664→   664→
   665→   665→  return {
   666→   666→    range: { from: fromD.toISOString(), to: toD.toISOString(), label },
   667→   667→    totals: {
   668→   668→      credit: creditPaise / 100,
   669→   669→      payments: paymentsPaise / 100,
   670→   670→      net: (creditPaise - paymentsPaise) / 100,
   671→   671→      creditPaise,
   672→   672→      paymentsPaise,
   673→   673→      count: countable.length,
   674→   674→    },
   675→   675→    series,
   676→   676→    customerOutstanding,
   677→   677→    recent: txns.slice(0, 10).map(publicTxn),
   678→   678→    meta: {
   679→   679→      customers: customers.length,
   680→   680→      withDues: customers.filter((c) => c.balancePaise > 0).length,
   681→   681→      settled: customers.filter((c) => c.balancePaise <= 0).length,
   682→   682→    },
   683→   683→  };
   684→   684→}

   762→   762→interface BackupCustomerInput {
   763→   763→  nameKey: string;
   764→   764→  name: string;
   765→   765→  phone?: string | null;
   766→   766→  notes?: string | null;

   769→   769→interface BackupTxnInput {
   770→   770→  id?: string;
   771→   771→  nameKey: string;
   772→   772→  name: string;
   773→   773→  amountPaise: number;
   774→   774→  txnType: string;
   775→   775→  rawText: string;
   776→   776→  note?: string | null;
   777→   777→  reversed?: boolean;
   778→   778→  isReversal?: boolean;

   781→   781→
   782→   782→export interface RestoreResult {
   783→   783→  mode: "replace" | "merge";
   784→   784→  importedCustomers: number;
   785→   785→  importedTransactions: number;
   786→   786→  skippedTransactions: number;
   787→   787→  exportedAt: string | null;
   788→   788→}
   789→   789→
   790→   790→const MAX_BACKUP_CUSTOMERS = 1000;
   791→   791→const MAX_BACKUP_TRANSACTIONS = 5000;
   792→   792→
   793→   793→/**
   794→   794→ * Restore a ledger from a backup produced by exportBackupJson().
   795→   795→ *
   796→   796→ * - "replace": wipe the current ledger (customers/transactions/reminders) and
   797→   797→ *   rebuild it from the backup. Audit history is KEPT and the import itself is
   798→   798→ *   recorded as a new audit event.
   799→   799→ * - "merge": add only what is missing (customers that don't exist yet,
   800→   800→ *   transactions whose id is new). Existing profile edits are never clobbered.
   801→   801→ *
   802→   802→ * Balances are ALWAYS recomputed from the imported ledger lines (never trusted
   803→   803→ * from the file), so the ledger stays internally consistent even if the backup
   804→   804→ * was truncated or hand-edited.
   805→   805→ */
   806→   806→export async function restoreBackup(
   807→   807→  backup: unknown,
   808→   808→  mode: "replace" | "merge"
   809→   809→): Promise<RestoreResult> {
   810→   810→  if (typeof backup !== "object" || backup === null) {
   811→   811→    throw new Error("The backup file does not look like a Udhaar Khata backup.");
   812→   812→  }
   813→   813→  const b = backup as Record<string, unknown>;
   814→   814→  if (b.format !== "udhaar-khata-backup") {
   815→   815→    throw new Error(
   816→   816→      "This file is not a Udhaar Khata backup (missing the format marker). Use the 'Full backup (JSON)' export."
   817→   817→    );
   818→   818→  }
   819→   819→  if (!Array.isArray(b.customers) || !Array.isArray(b.transactions)) {
   820→   820→    throw new Error("The backup is missing its customers or transactions list.");
   821→   821→  }
   822→   822→  if (b.customers.length > MAX_BACKUP_CUSTOMERS) {
   823→   823→    throw new Error(`Too many customers in the backup (max ${MAX_BACKUP_CUSTOMERS}).`);
   824→   824→  }
   825→   825→  if (b.transactions.length > MAX_BACKUP_TRANSACTIONS) {
   826→   826→    throw new Error(`Too many transactions in the backup (max ${MAX_BACKUP_TRANSACTIONS}).`);
   827→   827→  }
   828→   828→
   829→   829→  // ---- validate every row before touching the database ----
   830→   830→  const customers: BackupCustomerInput[] = [];
   831→   831→  for (let i = 0; i < b.customers.length; i++) {
   832→   832→    const c = b.customers[i] as Record<string, unknown>;
   833→   833→    const key = typeof c.nameKey === "string" ? c.nameKey.trim().toUpperCase() : "";
   834→   834→    const name = typeof c.name === "string" ? c.name.trim() : "";
   835→   835→    if (!key || !name) {
   836→   836→      throw new Error(`Customer row ${i + 1} is missing a name.`);
   837→   837→    }
   838→   838→    customers.push({
   839→   839→      nameKey: key,
   840→   840→      name,
   841→   841→      phone: typeof c.phone === "string" ? normalizePhone(c.phone) : null,
   842→   842→      notes: typeof c.notes === "string" ? c.notes : null,
   843→   843→    });
   844→   844→  }
   845→   845→
   846→   846→  const transactions: BackupTxnInput[] = [];
   847→   847→  for (let i = 0; i < b.transactions.length; i++) {
   848→   848→    const t = b.transactions[i] as Record<string, unknown>;
   849→   849→    const key = typeof t.nameKey === "string" ? t.nameKey.trim().toUpperCase() : "";
   850→   850→    const paise = typeof t.amountPaise === "number" ? t.amountPaise : NaN;
   851→   851→    const type = typeof t.txnType === "string" ? t.txnType : "";
   852→   852→    const at = typeof t.createdAt === "string" ? new Date(t.createdAt) : null;
   853→   853→    if (!key) throw new Error(`Transaction row ${i + 1} is missing the customer.`);
   854→   854→    if (!Number.isInteger(paise) || paise <= 0 || paise > RUPEE_MAX * 100) {
   855→   855→      throw new Error(`Transaction row ${i + 1} has an invalid amount.`);
   856→   856→    }
   857→   857→    if (type !== "UDHAAR" && type !== "PAYMENT") {
   858→   858→      throw new Error(`Transaction row ${i + 1} has an unknown entry type.`);
   859→   859→    }
   860→   860→    if (!at || Number.isNaN(at.getTime())) {
   861→   861→      throw new Error(`Transaction row ${i + 1} has an invalid date.`);
   862→   862→    }
   863→   863→    transactions.push({
   864→   864→      id: typeof t.id === "string" && t.id ? t.id : undefined,
   865→   865→      nameKey: key,
   866→   866→      name: typeof t.name === "string" && t.name.trim() ? t.name.trim() : key,
   867→   867→      amountPaise: paise,
   868→   868→      txnType: type,
   869→   869→      rawText:
   870→   870→        typeof t.rawText === "string" && t.rawText.trim()
   871→   871→          ? t.rawText
   872→   872→          : "Restored from backup",
   873→   873→      note: typeof t.note === "string" ? t.note : null,
   874→   874→      reversed: t.reversed === true,
   875→   875→      isReversal: t.isReversal === true,
   876→   876→      createdAt: at.toISOString(),
   877→   877→    });
   878→   878→  }
   879→   879→
   880→   880→  // Customer stubs for transactions whose customer row is missing from the
   881→   881→  // backup (keeps referential integrity without dropping ledger lines).
   882→   882→  const known = new Set(customers.map((c) => c.nameKey));
   883→   883→  for (const t of transactions) {
   884→   884→    if (!known.has(t.nameKey)) {
   885→   885→      customers.push({ nameKey: t.nameKey, name: t.name, phone: null, notes: null });
   886→   886→      known.add(t.nameKey);
   887→   887→    }
   888→   888→  }
   889→   889→
   890→   890→  const exportedAt = typeof b.exportedAt === "string" ? b.exportedAt : null;
   891→   891→
   892→   892→  const result = await db.$transaction(
   893→   893→    async (tx) => {
   894→   894→      let skipped = 0;
   895→   895→
   896→   896→      if (mode === "replace") {
   897→   897→        await tx.reminder.deleteMany();
   898→   898→        await tx.transaction.deleteMany();
   899→   899→        await tx.customer.deleteMany();
   900→   900→      }
   901→   901→
   902→   902→      // ---- customers ----
   903→   903→      if (mode === "merge") {
   904→   904→        // Merge never overwrites an existing customer's profile edits.
   905→   905→        const existing = new Set(
   906→   906→          (
   907→   907→            await tx.customer.findMany({
   908→   908→              where: { nameKey: { in: customers.map((c) => c.nameKey) } },
   909→   909→              select: { nameKey: true },
   910→   910→            })
   911→   911→          ).map((r) => r.nameKey)
   912→   912→        );
   913→   913→        const fresh = customers.filter((c) => !existing.has(c.nameKey));
   914→   914→        if (fresh.length > 0) {
   915→   915→          await tx.customer.createMany({
   916→   916→            data: fresh.map((c) => ({
   917→   917→              nameKey: c.nameKey,
   918→   918→              name: c.name,
   919→   919→              phone: c.phone,
   920→   920→              notes: c.notes,
   921→   921→            })),
   922→   922→          });
   923→   923→        }
   924→   924→      } else {
   925→   925→        await tx.customer.createMany({
   926→   926→          data: customers.map((c) => ({
   927→   927→            nameKey: c.nameKey,
   928→   928→            name: c.name,
   929→   929→            phone: c.phone,
   930→   930→            notes: c.notes,
   931→   931→          })),
   932→   932→        });
   933→   933→      }
   934→   934→
   935→   935→      // ---- transactions (dedupe by id in merge mode) ----
   936→   936→      let rows = transactions;
   937→   937→      if (mode === "merge" && transactions.some((t) => t.id)) {
   938→   938→        const ids = transactions
   939→   939→          .map((t) => t.id)
   940→   940→          .filter((x): x is string => Boolean(x));
   941→   941→        const existing = new Set(
   942→   942→          (
   943→   943→            await tx.transaction.findMany({
   944→   944→              where: { id: { in: ids } },
   945→   945→              select: { id: true },
   946→   946→            })
   947→   947→          ).map((r) => r.id)
   948→   948→        );
   949→   949→        rows = transactions.filter((t) => !t.id || !existing.has(t.id));
   950→   950→        skipped = transactions.length - rows.length;
   951→   951→      }
   952→   952→      if (rows.length > 0) {
   953→   953→        await tx.transaction.createMany({
   954→   954→          data: rows.map((t) => ({
   955→   955→            ...(t.id ? { id: t.id } : {}),
   956→   956→            nameKey: t.nameKey,
   957→   957→            name: t.name,
   958→   958→            amountPaise: t.amountPaise,
   959→   959→            txnType: t.txnType,
   960→   960→            rawText: t.rawText,
   961→   961→            note: t.note,
   962→   962→            reversed: t.reversed,
   963→   963→            isReversal: t.isReversal,
   964→   964→            createdAt: new Date(t.createdAt),
   965→   965→          })),
   966→   966→        });
   967→   967→      }
   968→   968→
   969→   969→      // ---- recompute every imported customer's balance FROM the ledger ----
   970→   970→      // (single aggregation pass — numbers in the backup are never trusted)
   971→   971→      const keys = [...new Set(mode === "replace" ? customers.map((c) => c.nameKey) : [...known])];
   972→   972→      const sums = await tx.transaction.groupBy({
   973→   973→        by: ["nameKey", "txnType"],
   974→   974→        where: { nameKey: { in: keys }, reversed: false, isReversal: false },
   975→   975→        _sum: { amountPaise: true },
   976→   976→      });
   977→   977→      const ranges = await tx.transaction.groupBy({
   978→   978→        by: ["nameKey"],
   979→   979→        where: { nameKey: { in: keys }, reversed: false, isReversal: false },
   980→   980→        _min: { createdAt: true },
   981→   981→        _max: { createdAt: true },
   982→   982→      });
   983→   983→      const firstDues = await tx.transaction.groupBy({
   984→   984→        by: ["nameKey"],
   985→   985→        where: { nameKey: { in: keys }, txnType: "UDHAAR", reversed: false, isReversal: false },
   986→   986→        _min: { createdAt: true },
   987→   987→      });
   988→   988→
   989→   989→      const balanceOf = new Map<string, number>();
   990→   990→      for (const s of sums) {
   991→   991→        const delta = s.txnType === "UDHAAR" ? s._sum.amountPaise : -s._sum.amountPaise;
   992→   992→        balanceOf.set(s.nameKey, (balanceOf.get(s.nameKey) ?? 0) + delta);
   993→   993→      }
   994→   994→      const lastOf = new Map(ranges.map((r) => [r.nameKey, r._max.createdAt]));
   995→   995→      const dueOf = new Map(firstDues.map((r) => [r.nameKey, r._min.createdAt]));
   996→   996→
   997→   997→      for (const key of keys) {
   998→   998→        const balance = balanceOf.get(key) ?? 0;
   999→   999→        await tx.customer.update({
  1000→  1000→          where: { nameKey: key },
  1001→  1001→          data: {
  1002→  1002→            balancePaise: balance,
  1003→  1003→            lastTxnAt: lastOf.get(key) ?? null,
  1004→  1004→            firstDueAt: balance > 0 ? (dueOf.get(key) ?? null) : null,
  1005→  1005→          },
  1006→  1006→        });
  1007→  1007→      }
  1008→  1008→
  1009→  1009→      // A restore of real data means the ledger is no longer "demo".
  1010→  1010→      // (tx-scoped — the global db client must not be touched while the
  1011→  1011→      // SQLite transaction is open, it deadlocks the database file.)
  1012→  1012→      await tx.settings.upsert({
  1013→  1013→        where: { id: "singleton" },
  1014→  1014→        create: { id: "singleton" },
  1015→  1015→        update: {},
  1016→  1016→      });
  1017→  1017→      if (mode === "replace") {
  1018→  1018→        await tx.settings.update({
  1019→  1019→          where: { id: "singleton" },
  1020→  1020→          data: { demoData: false },
  1021→  1021→        });
  1022→  1022→      }
  1023→  1023→
  1024→  1024→      await tx.auditEvent.create({
  1025→  1025→        data: {
  1026→  1026→          action: "import_backup",
  1027→  1027→          entity: "settings",
  1028→  1028→          entityId: "singleton",
  1029→  1029→          data: JSON.stringify({
  1030→  1030→            mode,
  1031→  1031→            importedCustomers: customers.length,
  1032→  1032→            importedTransactions: transactions.length - skipped,
  1033→  1033→            skippedTransactions: skipped,
  1034→  1034→            exportedAt,
  1035→  1035→          }),
  1036→  1036→        },
  1037→  1037→      });
  1038→  1038→
  1039→  1039→      return { skipped };
  1040→  1040→    },
  1041→  1041→    { timeout: 30_000, maxWait: 10_000 }
  1042→  1042→  );
  1043→  1043→
  1044→  1044→  return {
  1045→  1045→    mode,
  1046→  1046→    importedCustomers: customers.length,
  1047→  1047→    importedTransactions: transactions.length - result.skipped,
  1048→  1048→    skippedTransactions: result.skipped,
  1049→  1049→    exportedAt,
  1050→  1050→  };
  1051→  1051→}
