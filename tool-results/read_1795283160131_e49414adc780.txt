    23→/* ---------- minimal Web Speech API types ---------- */
    24→interface SpeechRecognitionAlternative {
    25→  transcript: string;

    27→interface SpeechRecognitionResult {
    28→  0: SpeechRecognitionAlternative;

    31→interface SpeechRecognitionResultList {
    32→  0: SpeechRecognitionResult;

    34→interface SpeechRecognitionEvent extends Event {
    35→  results: SpeechRecognitionResultList;

    37→interface SpeechRecognition extends EventTarget {
    38→  lang: string;
    39→  interimResults: boolean;
    40→  maxAlternatives: number;
    41→  start(): void;
    42→  stop(): void;
    43→  onresult: ((e: SpeechRecognitionEvent) => void) | null;
    44→  onerror: ((e: Event) => void) | null;
    45→  onend: (() => void) | null;
    46→}
    47→type SpeechRecognitionCtor = new () => SpeechRecognition;

    93→function timeAgo(iso: string | null): string {
    94→  if (!iso) return "";
    95→  const d = new Date(iso);
    96→  const mins = Math.floor((Date.now() - d.getTime()) / 60000);
    97→  if (mins < 1) return "just now";
    98→  if (mins < 60) return `${mins} min ago`;
    99→  const hrs = Math.floor(mins / 60);
   100→  if (hrs < 24) return `${hrs} hr ago`;
   101→  const days = Math.floor(hrs / 24);
   102→  if (days < 30) return `${days} day${days === 1 ? "" : "s"} ago`;
   103→  return d.toLocaleDateString("en-IN", { day: "numeric", month: "short" });
   104→}
   105→
   106→function daysOverdue(iso: string | null): number {
   107→  if (!iso) return 0;
   108→  const delta = Date.now() - new Date(iso).getTime();
   109→  return Math.max(0, Math.floor(delta / 86_400_000));
   110→}
