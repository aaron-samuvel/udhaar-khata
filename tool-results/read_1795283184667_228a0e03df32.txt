    93→    93→function timeAgo(iso: string | null): string {
    94→    94→  if (!iso) return "";
    95→    95→  const d = new Date(iso);
    96→    96→  const mins = Math.floor((Date.now() - d.getTime()) / 60000);
    97→    97→  if (mins < 1) return "just now";
    98→    98→  if (mins < 60) return `${mins} min ago`;
    99→    99→  const hrs = Math.floor(mins / 60);
   100→   100→  if (hrs < 24) return `${hrs} hr ago`;
   101→   101→  const days = Math.floor(hrs / 24);
   102→   102→  if (days < 30) return `${days} day${days === 1 ? "" : "s"} ago`;
   103→   103→  return d.toLocaleDateString("en-IN", { day: "numeric", month: "short" });
   104→   104→}
   105→   105→
   106→   106→function daysOverdue(iso: string | null): number {
   107→   107→  if (!iso) return 0;
   108→   108→  const delta = Date.now() - new Date(iso).getTime();
   109→   109→  return Math.max(0, Math.floor(delta / 86_400_000));
   110→   110→}
