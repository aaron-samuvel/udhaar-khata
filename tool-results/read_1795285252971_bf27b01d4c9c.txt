    24→/** Keep only 10-digit Indian mobile numbers (strips +91 / 91 prefixes). */
    25→export function normalizePhone(phone?: string | null): string | null {
    26→  if (!phone) return null;
    27→  let digits = String(phone).replace(/\D/g, "");
    28→  if (digits.startsWith("91") && digits.length === 12) digits = digits.slice(2);
    29→  if (digits.length === 10) return digits;
    30→  return null;
    31→}

    33→/** Render the reminder template. Supported placeholders: {name} {amount} {shop} {days}. */
    34→export function renderTemplate(
    35→  tpl: string,
    36→  vars: { name: string; amount: string; shop: string; days: number }
    37→): string {
    38→  return tpl
    39→    .replace(/\{name\}/g, vars.name)
    40→    .replace(/\{amount\}/g, vars.amount)
    41→    .replace(/\{shop\}/g, vars.shop)
    42→    .replace(/\{days\}/g, String(vars.days));
    43→}

   157→export interface AddTxnInput {
   158→  customer: string;
   159→  amountPaise: number;
   160→  txnType: TxnType;
   161→  rawText: string;
   162→  note?: string | null;
   163→  phone?: string | null;

   165→  force?: boolean;
   166→  isReversal?: boolean;

   170→export async function addTransaction(
   171→  input: AddTxnInput
   172→): Promise<{ transaction: TxnPublic; customer: CustomerPublic; duplicate?: TxnPublic }> {
   173→  if (!Number.isInteger(input.amountPaise) || input.amountPaise <= 0) {
   174→    throw new Error("Amount must be greater than zero.");
   175→  }
   176→  const key = nameKey(input.customer);
   177→  if (!key) throw new Error("Customer name is required.");
   178→  const phone = normalizePhone(input.phone);
   179→  if (input.phone && !phone) {
   180→    throw new Error("Phone must be a 10-digit Indian mobile number (with optional +91).");
   181→  }
   182→
   183→  const existing = await db.customer.findUnique({ where: { nameKey: key } });
   184→
   185→  if (!input.force && !input.isReversal) {
   186→    const dupe = await db.transaction.findFirst({
   187→      where: {
   188→        nameKey: key,
   189→        txnType: input.txnType,
   190→        amountPaise: input.amountPaise,
   191→        reversed: false,
   192→        isReversal: false,
   193→        createdAt: { gte: new Date(Date.now() - DUPLICATE_WINDOW_MS) },
   194→      },
   195→      orderBy: { createdAt: "desc" },
   196→    });
   197→    if (dupe) {
   198→      return {
   199→        transaction: publicTxn(dupe),
   200→        customer: existing ? publicCustomer(existing) : publicTxnAsCustomer(dupe),
   201→        duplicate: publicTxn(dupe),
   202→      };
   203→    }
   204→  }
   205→
   206→  const now = new Date();
   207→  const delta = input.txnType === "UDHAAR" ? input.amountPaise : -input.amountPaise;
   208→  const oldBalance = existing?.balancePaise ?? 0;
   209→  const newBalance = oldBalance + delta;
   210→
   211→  let firstDueAt = existing?.firstDueAt ?? null;
   212→  if (newBalance > 0 && (!firstDueAt || oldBalance <= 0)) firstDueAt = now;
   213→  if (newBalance <= 0) firstDueAt = null;
   214→
   215→  const displayName = existing?.name ?? input.customer.trim();
   216→  const savedPhone = phone ?? existing?.phone ?? null;
   217→
   218→  const txn = await db.$transaction(async (tx) => {
   219→    await tx.customer.upsert({
   220→      where: { nameKey: key },
   221→      create: {
   222→        nameKey: key,
   223→        name: displayName,
   224→        balancePaise: newBalance,
   225→        phone: savedPhone,
   226→        firstDueAt,
   227→        lastTxnAt: now,
   228→      },
   229→      update: {
   230→        name: displayName,
   231→        balancePaise: newBalance,
   232→        phone: savedPhone,
   233→        firstDueAt,
   234→        lastTxnAt: now,
   235→      },
   236→    });
   237→    const created = await tx.transaction.create({
   238→      data: {
   239→        nameKey: key,
   240→        name: displayName,
   241→        amountPaise: input.amountPaise,
   242→        txnType: input.txnType,
   243→        rawText: input.rawText,
   244→        note: input.note ?? null,
   245→        isReversal: input.isReversal ?? false,
   246→        reversedById: input.reversedById ?? null,
   247→        createdAt: now,
   248→      },
   249→    });
   250→    await tx.auditEvent.create({
   251→      data: {
   252→        action: "transaction_create",
   253→        entity: "transaction",
   254→        entityId: created.id,
   255→        data: JSON.stringify({
   256→          customer: displayName,
   257→          txnType: input.txnType,
   258→          amountPaise: input.amountPaise,
   259→          forced: input.force ?? false,
   260→          isReversal: input.isReversal ?? false,
   261→        }),
   262→      },
   263→    });
   264→    return created;
   265→  });
   266→
   267→  const cust = (await db.customer.findUnique({ where: { nameKey: key } }))!;
   268→  return { transaction: publicTxn(txn), customer: publicCustomer(cust) };
   269→}

   285→/**
   286→ * Reverse a ledger line by creating an offsetting entry. The original is
   287→ * flagged (never deleted or mutated in place) so the audit history survives.
   288→ * Reversal lines are excluded from report totals — no money actually moved.
   289→ */
   290→export async function reverseTransaction(id: string) {
   291→  const result = await db.$transaction(async (tx) => {
   292→    const orig = await tx.transaction.findUnique({ where: { id } });
   293→    if (!orig) throw new Error("Transaction not found.");
   294→    if (orig.reversed) throw new Error("This transaction has already been reversed.");
   295→    if (orig.isReversal) {
   296→      throw new Error("A reversal entry cannot be reversed. Reverse the original transaction instead.");
   297→    }
   298→    const customer = await tx.customer.findUnique({ where: { nameKey: orig.nameKey } });
   299→    if (!customer) throw new Error("Customer record missing for this transaction.");
   300→
   301→    const now = new Date();
   302→    const flip: TxnType = orig.txnType === "UDHAAR" ? "PAYMENT" : "UDHAAR";
   303→    const delta = flip === "UDHAAR" ? orig.amountPaise : -orig.amountPaise;
   304→    const newBalance = customer.balancePaise + delta;
   305→
   306→    let firstDueAt = customer.firstDueAt;
   307→    if (newBalance > 0 && (!firstDueAt || customer.balancePaise <= 0)) firstDueAt = now;
   308→    if (newBalance <= 0) firstDueAt = null;
   309→
   310→    const reversal = await tx.transaction.create({
   311→      data: {
   312→        nameKey: orig.nameKey,
   313→        name: orig.name,
   314→        amountPaise: orig.amountPaise,
   315→        txnType: flip,
   316→        rawText: `Reversal of: ${orig.rawText}`,
   317→        note: `Reversal of transaction ${orig.id}`,
   318→        isReversal: true,
   319→        reversedById: orig.id,
   320→        createdAt: now,
   321→      },
   322→    });
   323→    await tx.transaction.update({
   324→      where: { id: orig.id },
   325→      data: { reversed: true, reversedById: reversal.id },
   326→    });
   327→    const updated = await tx.customer.update({
   328→      where: { nameKey: orig.nameKey },
   329→      data: { balancePaise: newBalance, firstDueAt, lastTxnAt: now },
   330→    });
   331→    await tx.auditEvent.create({
   332→      data: {
   333→        action: "transaction_reverse",
   334→        entity: "transaction",
   335→        entityId: orig.id,
   336→        data: JSON.stringify({
   337→          reversalId: reversal.id,
   338→          customer: orig.name,
   339→          amountPaise: orig.amountPaise,
   340→          originalType: orig.txnType,
   341→        }),
   342→      },
   343→    });
   344→    return { reversal, customer: updated };
   345→  });
   346→  return {
   347→    transaction: publicTxn(result.reversal),
   348→    customer: publicCustomer(result.customer),
   349→  };
   350→}

   375→export async function updateCustomer(
   376→  name: string,
   377→  patch: { phone?: string | null; notes?: string | null }
   378→): Promise<CustomerPublic | null> {
   379→  const key = nameKey(name);
   380→  const existing = await db.customer.findUnique({ where: { nameKey: key } });
   381→  if (!existing) return null;
   382→
   383→  const data: { phone?: string | null; notes?: string | null } = {};
   384→  if (patch.phone !== undefined) {
   385→    if (patch.phone === null || patch.phone === "") {
   386→      data.phone = null;
   387→    } else {
   388→      const p = normalizePhone(patch.phone);
   389→      if (!p) throw new Error("Phone must be a 10-digit Indian mobile number (with optional +91).");
   390→      data.phone = p;
   391→    }
   392→  }
   393→  if (patch.notes !== undefined) {
   394→    const notes = patch.notes === null ? null : patch.notes.trim();
   395→    data.notes = notes === "" ? null : notes;
   396→  }
   397→
   398→  const updated = await db.$transaction(async (tx) => {
   399→    const row = await tx.customer.update({ where: { nameKey: key }, data });
   400→    await tx.auditEvent.create({
   401→      data: {
   402→        action: "update_customer",
   403→        entity: "customer",
   404→        entityId: key,
   405→        data: JSON.stringify({
   406→          phoneChanged: patch.phone !== undefined,
   407→          notesChanged: patch.notes !== undefined,
   408→        }),
   409→      },
   410→    });
   411→    return row;
   412→  });
   413→  return publicCustomer(updated);
   414→}

   448→/**
   449→ * Reminder run (the daily-job equivalent): find customers whose dues are
   450→ * older than `overdueDays` (default from Settings) and write reminder records
   451→ * with ready WhatsApp links.
   452→ */
   453→export async function writeReminders(overdueDays?: number): Promise<ReminderPublic[]> {
   454→  const settings = await getSettings();
   455→  const days = Math.max(0, Math.floor(overdueDays ?? settings.overdueDays));
   456→  const rid = Math.random().toString(36).slice(2, 12);
   457→  const written: ReminderPublic[] = [];
   458→
   459→  for (const c of await db.customer.findMany()) {
   460→    if (c.balancePaise <= 0) continue;
   461→    const dueFrom = c.firstDueAt ?? c.lastTxnAt;
   462→    const age = daysSince(dueFrom);
   463→    if (age < days) continue;
   464→
   465→    const { url, message } = await whatsappLink(c.name, c.balancePaise, c.phone, age);
   466→    const row = await db.reminder.create({
   467→      data: {
   468→        nameKey: c.nameKey,
   469→        name: c.name,
   470→        balancePaise: c.balancePaise,
   471→        daysOverdue: age,
   472→        phone: c.phone,
   473→        whatsappUrl: url,
   474→        message,
   475→        runId: rid,
   476→      },
   477→    });
   478→    written.push(publicReminder(row));
   479→  }
   480→  return written;
   481→}

   506→/**
   507→ * WhatsApp-ready STATEMENT for one customer: their last ledger lines (real
   508→ * data only), the outstanding balance and the shop name. Unlike a reminder,
   509→ * a statement works for settled customers too — useful as a receipt.
   510→ */
   511→export async function customerStatement(
   512→  name: string
   513→): Promise<{
   514→  whatsappUrl: string;
   515→  message: string;
   516→  customer: CustomerPublic;
   517→  lines: number;
   518→} | null> {
   519→  const key = nameKey(name);
   520→  const c = await db.customer.findUnique({ where: { nameKey: key } });
   521→  if (!c) return null;
   522→
   523→  const txns = await db.transaction.findMany({
   524→    where: { nameKey: key, reversed: false, isReversal: false },
   525→    orderBy: { createdAt: "desc" },
   526→    take: 20,
   527→  });
   528→  const settings = await getSettings();
   529→
   530→  const lines = txns.map((t) => {
   531→    const sign = t.txnType === "UDHAAR" ? "+" : "−";
   532→    return `${istTimestamp(t.createdAt)}  ${sign}₹${(t.amountPaise / 100).toFixed(2)} — ${t.rawText}`;
   533→  });
   534→
   535→  const balanceLine =
   536→    c.balancePaise > 0
   537→      ? `Udhaar pending: ₹${(c.balancePaise / 100).toFixed(2)}`
   538→      : `Khata settled: ₹${(c.balancePaise / 100).toFixed(2)}`;
   539→
   540→  const message = [
   541→    `*${settings.shopName}* — khata statement`,
   542→    `Customer: ${c.name}`,
   543→    "",
   544→    ...(lines.length > 0 ? lines : ["(no entries yet)"]),
   545→    "",
   546→    balanceLine,
   547→    c.balancePaise > 0
   548→      ? "Kripya jald payment kar dein. Dhanyavaad!"
   549→      : "Payment ke liye dhanyavaad!",
   550→  ].join("\n");
   551→
   552→  const encoded = encodeURIComponent(message);
   553→  const digits = normalizePhone(c.phone);
   554→  const whatsappUrl = digits
   555→    ? `https://wa.me/91${digits}?text=${encoded}`
   556→    : `https://wa.me/?text=${encoded}`;
   557→
   558→  return { whatsappUrl, message, customer: publicCustomer(c), lines: lines.length };
   559→}

   562→
   563→export type ReportRange = "today" | "7d" | "30d" | "all" | "custom";
   564→
   565→/**
   566→ * Reports computed from actual stored transactions.
   567→ *
   568→ * Counting rules (documented, deterministic, no double-counting):
   569→ * - Lines with `reversed=true` are voided → excluded from totals.
   570→ * - Lines with `isReversal=true` are corrections, not real money movement →
   571→ *   excluded from totals (but visible in recent activity).
   572→ * - Current outstanding comes from Customer.balancePaise (source of truth).
   573→ * - Day boundaries are in IST (Asia/Kolkata).
   574→ */
   575→export async function getReports(
   576→  range: ReportRange,
   577→  fromStr?: string,
   578→  toStr?: string
   579→) {
   580→  const now = new Date();
   581→  let fromD: Date;
   582→  let toD = now;
   583→  let label: string;
   584→
   585→  switch (range) {
   586→    case "today":
   587→      fromD = istDayStart(now);
   588→      label = "Today";
   589→      break;
   590→    case "7d":
   591→      fromD = istDayStart(new Date(now.getTime() - 6 * 86_400_000));
   592→      label = "Last 7 days";
   593→      break;
   594→    case "30d":
   595→      fromD = istDayStart(new Date(now.getTime() - 29 * 86_400_000));
   596→      label = "Last 30 days";
   597→      break;
   598→    case "custom": {
   599→      if (!fromStr || Number.isNaN(new Date(fromStr).getTime())) {
   600→        throw new Error("A valid 'from' date is required for a custom range.");
   601→      }
   602→      fromD = new Date(fromStr);
   603→      if (toStr && !Number.isNaN(new Date(toStr).getTime())) toD = new Date(toStr);
   604→      label = "Custom range";
   605→      break;
   606→    }
   607→    case "all":
   608→    default:
   609→      fromD = new Date(0);
   610→      label = "All time";
   611→      break;
   612→  }
   613→
   614→  const txns = await db.transaction.findMany({
   615→    where: { createdAt: { gte: fromD, lte: toD } },
   616→    orderBy: { createdAt: "desc" },
   617→  });
   618→  const countable = txns.filter((t) => !t.reversed && !t.isReversal);
   619→
   620→  let creditPaise = 0;
   621→  let paymentsPaise = 0;
   622→  const seriesMap = new Map<string, { credit: number; payments: number }>();
   623→  for (const t of countable) {
   624→    const key = istDayKey(t.createdAt);
   625→    const bucket = seriesMap.get(key) ?? { credit: 0, payments: 0 };
   626→    if (t.txnType === "UDHAAR") {
   627→      creditPaise += t.amountPaise;
   628→      bucket.credit += t.amountPaise;
   629→    } else {
   630→      paymentsPaise += t.amountPaise;
   631→      bucket.payments += t.amountPaise;
   632→    }
   633→    seriesMap.set(key, bucket);
   634→  }
   635→
   636→  // Fill day buckets for short ranges so charts show continuous days.
   637→  const series: { date: string; credit: number; payments: number }[] = [];
   638→  const fillFrom = new Date(Math.max(fromD.getTime(), now.getTime() - 30 * 86_400_000));
   639→  if (range !== "all") {
   640→    for (let d = istDayStart(fillFrom); d.getTime() <= toD.getTime(); d = new Date(d.getTime() + 86_400_000)) {
   641→      const key = istDayKey(d);
   642→      const b = seriesMap.get(key) ?? { credit: 0, payments: 0 };
   643→      series.push({ date: key, credit: b.credit / 100, payments: b.payments / 100 });
   644→    }
   645→  } else {
   646→    for (const key of [...seriesMap.keys()].sort()) {
   647→      const b = seriesMap.get(key)!;
   648→      series.push({ date: key, credit: b.credit / 100, payments: b.payments / 100 });
   649→    }
   650→  }
   651→
   652→  const customers = await db.customer.findMany();
   653→  const customerOutstanding = customers
   654→    .filter((c) => c.balancePaise > 0)
   655→    .map((c) => ({
   656→      name: c.name,
   657→      phone: c.phone,
   658→      balance: c.balancePaise / 100,
   659→      balancePaise: c.balancePaise,
   660→      daysDue: daysSince(c.firstDueAt ?? c.lastTxnAt),
   661→      notes: c.notes,
   662→    }))
   663→    .sort((a, b) => b.balancePaise - a.balancePaise);
   664→
   665→  return {
   666→    range: { from: fromD.toISOString(), to: toD.toISOString(), label },
   667→    totals: {
   668→      credit: creditPaise / 100,
   669→      payments: paymentsPaise / 100,
   670→      net: (creditPaise - paymentsPaise) / 100,
   671→      creditPaise,
   672→      paymentsPaise,
   673→      count: countable.length,
   674→    },
   675→    series,
   676→    customerOutstanding,
   677→    recent: txns.slice(0, 10).map(publicTxn),
   678→    meta: {
   679→      customers: customers.length,
   680→      withDues: customers.filter((c) => c.balancePaise > 0).length,
   681→      settled: customers.filter((c) => c.balancePaise <= 0).length,
   682→    },
   683→  };
   684→}

   761→
   762→interface BackupCustomerInput {
   763→  nameKey: string;
   764→  name: string;
   765→  phone?: string | null;
   766→  notes?: string | null;
   767→}
   768→
   769→interface BackupTxnInput {
   770→  id?: string;
   771→  nameKey: string;
   772→  name: string;
   773→  amountPaise: number;
   774→  txnType: string;
   775→  rawText: string;
   776→  note?: string | null;
   777→  reversed?: boolean;
   778→  isReversal?: boolean;
   779→  createdAt: string;
   780→}
   781→
   782→export interface RestoreResult {
   783→  mode: "replace" | "merge";
   784→  importedCustomers: number;
   785→  importedTransactions: number;
   786→  skippedTransactions: number;
   787→  exportedAt: string | null;
   788→}
   789→
   790→const MAX_BACKUP_CUSTOMERS = 1000;
   791→const MAX_BACKUP_TRANSACTIONS = 5000;
   792→
   793→/**
   794→ * Restore a ledger from a backup produced by exportBackupJson().
   795→ *
   796→ * - "replace": wipe the current ledger (customers/transactions/reminders) and
   797→ *   rebuild it from the backup. Audit history is KEPT and the import itself is
   798→ *   recorded as a new audit event.
   799→ * - "merge": add only what is missing (customers that don't exist yet,
   800→ *   transactions whose id is new). Existing profile edits are never clobbered.
   801→ *
   802→ * Balances are ALWAYS recomputed from the imported ledger lines (never trusted
   803→ * from the file), so the ledger stays internally consistent even if the backup
   804→ * was truncated or hand-edited.
   805→ */
   806→export async function restoreBackup(
   807→  backup: unknown,
   808→  mode: "replace" | "merge"
   809→): Promise<RestoreResult> {
   810→  if (typeof backup !== "object" || backup === null) {
   811→    throw new Error("The backup file does not look like a Udhaar Khata backup.");
   812→  }
   813→  const b = backup as Record<string, unknown>;
   814→  if (b.format !== "udhaar-khata-backup") {
   815→    throw new Error(
   816→      "This file is not a Udhaar Khata backup (missing the format marker). Use the 'Full backup (JSON)' export."
   817→    );
   818→  }
   819→  if (!Array.isArray(b.customers) || !Array.isArray(b.transactions)) {
   820→    throw new Error("The backup is missing its customers or transactions list.");
   821→  }
   822→  if (b.customers.length > MAX_BACKUP_CUSTOMERS) {
   823→    throw new Error(`Too many customers in the backup (max ${MAX_BACKUP_CUSTOMERS}).`);
   824→  }
   825→  if (b.transactions.length > MAX_BACKUP_TRANSACTIONS) {
   826→    throw new Error(`Too many transactions in the backup (max ${MAX_BACKUP_TRANSACTIONS}).`);
   827→  }
   828→
   829→  // ---- validate every row before touching the database ----
   830→  const customers: BackupCustomerInput[] = [];
   831→  for (let i = 0; i < b.customers.length; i++) {
   832→    const c = b.customers[i] as Record<string, unknown>;
   833→    const key = typeof c.nameKey === "string" ? c.nameKey.trim().toUpperCase() : "";
   834→    const name = typeof c.name === "string" ? c.name.trim() : "";
   835→    if (!key || !name) {
   836→      throw new Error(`Customer row ${i + 1} is missing a name.`);
   837→    }
   838→    customers.push({
   839→      nameKey: key,
   840→      name,
   841→      phone: typeof c.phone === "string" ? normalizePhone(c.phone) : null,
   842→      notes: typeof c.notes === "string" ? c.notes : null,
   843→    });
   844→  }
   845→
   846→  const transactions: BackupTxnInput[] = [];
   847→  for (let i = 0; i < b.transactions.length; i++) {
   848→    const t = b.transactions[i] as Record<string, unknown>;
   849→    const key = typeof t.nameKey === "string" ? t.nameKey.trim().toUpperCase() : "";
   850→    const paise = typeof t.amountPaise === "number" ? t.amountPaise : NaN;
   851→    const type = typeof t.txnType === "string" ? t.txnType : "";
   852→    const at = typeof t.createdAt === "string" ? new Date(t.createdAt) : null;
   853→    if (!key) throw new Error(`Transaction row ${i + 1} is missing the customer.`);
   854→    if (!Number.isInteger(paise) || paise <= 0 || paise > RUPEE_MAX * 100) {
   855→      throw new Error(`Transaction row ${i + 1} has an invalid amount.`);
   856→    }
   857→    if (type !== "UDHAAR" && type !== "PAYMENT") {
   858→      throw new Error(`Transaction row ${i + 1} has an unknown entry type.`);
   859→    }
   860→    if (!at || Number.isNaN(at.getTime())) {
   861→      throw new Error(`Transaction row ${i + 1} has an invalid date.`);
   862→    }
   863→    transactions.push({
   864→      id: typeof t.id === "string" && t.id ? t.id : undefined,
   865→      nameKey: key,
   866→      name: typeof t.name === "string" && t.name.trim() ? t.name.trim() : key,
   867→      amountPaise: paise,
   868→      txnType: type,
   869→      rawText:
   870→        typeof t.rawText === "string" && t.rawText.trim()
   871→          ? t.rawText
   872→          : "Restored from backup",
   873→      note: typeof t.note === "string" ? t.note : null,
   874→      reversed: t.reversed === true,
   875→      isReversal: t.isReversal === true,
   876→      createdAt: at.toISOString(),
   877→    });
   878→  }
   879→
   880→  // Customer stubs for transactions whose customer row is missing from the
   881→  // backup (keeps referential integrity without dropping ledger lines).
   882→  const known = new Set(customers.map((c) => c.nameKey));
   883→  for (const t of transactions) {
   884→    if (!known.has(t.nameKey)) {
   885→      customers.push({ nameKey: t.nameKey, name: t.name, phone: null, notes: null });
   886→      known.add(t.nameKey);
   887→    }
   888→  }
   889→
   890→  const exportedAt = typeof b.exportedAt === "string" ? b.exportedAt : null;
   891→
   892→  const result = await db.$transaction(
   893→    async (tx) => {
   894→      let skipped = 0;
   895→
   896→      if (mode === "replace") {
   897→        await tx.reminder.deleteMany();
   898→        await tx.transaction.deleteMany();
   899→        await tx.customer.deleteMany();
   900→      }
   901→
   902→      // ---- customers ----
   903→      if (mode === "merge") {
   904→        // Merge never overwrites an existing customer's profile edits.
   905→        const existing = new Set(
   906→          (
   907→            await tx.customer.findMany({
   908→              where: { nameKey: { in: customers.map((c) => c.nameKey) } },
   909→              select: { nameKey: true },
   910→            })
   911→          ).map((r) => r.nameKey)
   912→        );
   913→        const fresh = customers.filter((c) => !existing.has(c.nameKey));
   914→        if (fresh.length > 0) {
   915→          await tx.customer.createMany({
   916→            data: fresh.map((c) => ({
   917→              nameKey: c.nameKey,
   918→              name: c.name,
   919→              phone: c.phone,
   920→              notes: c.notes,
   921→            })),
   922→          });
   923→        }
   924→      } else {
   925→        await tx.customer.createMany({
   926→          data: customers.map((c) => ({
   927→            nameKey: c.nameKey,
   928→            name: c.name,
   929→            phone: c.phone,
   930→            notes: c.notes,
   931→          })),
   932→        });
   933→      }
   934→
   935→      // ---- transactions (dedupe by id in merge mode) ----
   936→      let rows = transactions;
   937→      if (mode === "merge" && transactions.some((t) => t.id)) {
   938→        const ids = transactions
   939→          .map((t) => t.id)
   940→          .filter((x): x is string => Boolean(x));
   941→        const existing = new Set(
   942→          (
   943→            await tx.transaction.findMany({
   944→              where: { id: { in: ids } },
   945→              select: { id: true },
   946→            })
   947→          ).map((r) => r.id)
   948→        );
   949→        rows = transactions.filter((t) => !t.id || !existing.has(t.id));
   950→        skipped = transactions.length - rows.length;
   951→      }
   952→      if (rows.length > 0) {
   953→        await tx.transaction.createMany({
   954→          data: rows.map((t) => ({
   955→            ...(t.id ? { id: t.id } : {}),
   956→            nameKey: t.nameKey,
   957→            name: t.name,
   958→            amountPaise: t.amountPaise,
   959→            txnType: t.txnType,
   960→            rawText: t.rawText,
   961→            note: t.note,
   962→            reversed: t.reversed,
   963→            isReversal: t.isReversal,
   964→            createdAt: new Date(t.createdAt),
   965→          })),
   966→        });
   967→      }
   968→
   969→      // ---- recompute every imported customer's balance FROM the ledger ----
   970→      // (single aggregation pass — numbers in the backup are never trusted)
   971→      const keys = [...new Set(mode === "replace" ? customers.map((c) => c.nameKey) : [...known])];
   972→      const sums = await tx.transaction.groupBy({
   973→        by: ["nameKey", "txnType"],
   974→        where: { nameKey: { in: keys }, reversed: false, isReversal: false },
   975→        _sum: { amountPaise: true },
   976→      });
   977→      const ranges = await tx.transaction.groupBy({
   978→        by: ["nameKey"],
   979→        where: { nameKey: { in: keys }, reversed: false, isReversal: false },
   980→        _min: { createdAt: true },
   981→        _max: { createdAt: true },
   982→      });
   983→      const firstDues = await tx.transaction.groupBy({
   984→        by: ["nameKey"],
   985→        where: { nameKey: { in: keys }, txnType: "UDHAAR", reversed: false, isReversal: false },
   986→        _min: { createdAt: true },
   987→      });
   988→
   989→      const balanceOf = new Map<string, number>();
   990→      for (const s of sums) {
   991→        const delta = s.txnType === "UDHAAR" ? s._sum.amountPaise : -s._sum.amountPaise;
   992→        balanceOf.set(s.nameKey, (balanceOf.get(s.nameKey) ?? 0) + delta);
   993→      }
   994→      const lastOf = new Map(ranges.map((r) => [r.nameKey, r._max.createdAt]));
   995→      const dueOf = new Map(firstDues.map((r) => [r.nameKey, r._min.createdAt]));
   996→
   997→      for (const key of keys) {
   998→        const balance = balanceOf.get(key) ?? 0;
   999→        await tx.customer.update({
  1000→          where: { nameKey: key },
  1001→          data: {
  1002→            balancePaise: balance,
  1003→            lastTxnAt: lastOf.get(key) ?? null,
  1004→            firstDueAt: balance > 0 ? (dueOf.get(key) ?? null) : null,
  1005→          },
  1006→        });
  1007→      }
  1008→
  1009→      // A restore of real data means the ledger is no longer "demo".
  1010→      // (tx-scoped — the global db client must not be touched while the
  1011→      // SQLite transaction is open, it deadlocks the database file.)
  1012→      await tx.settings.upsert({
  1013→        where: { id: "singleton" },
  1014→        create: { id: "singleton" },
  1015→        update: {},
  1016→      });
  1017→      if (mode === "replace") {
  1018→        await tx.settings.update({
  1019→          where: { id: "singleton" },
  1020→          data: { demoData: false },
  1021→        });
  1022→      }
  1023→
  1024→      await tx.auditEvent.create({
  1025→        data: {
  1026→          action: "import_backup",
  1027→          entity: "settings",
  1028→          entityId: "singleton",
  1029→          data: JSON.stringify({
  1030→            mode,
  1031→            importedCustomers: customers.length,
  1032→            importedTransactions: transactions.length - skipped,
  1033→            skippedTransactions: skipped,
  1034→            exportedAt,
  1035→          }),
  1036→        },
  1037→      });
  1038→
  1039→      return { skipped };
  1040→    },
  1041→    { timeout: 30_000, maxWait: 10_000 }
  1042→  );
  1043→
  1044→  return {
  1045→    mode,
  1046→    importedCustomers: customers.length,
  1047→    importedTransactions: transactions.length - result.skipped,
  1048→    skippedTransactions: result.skipped,
  1049→    exportedAt,
  1050→  };
  1051→}
